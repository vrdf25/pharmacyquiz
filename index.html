// ===================== QUIZRX BACKEND v2 ===================== // Sheet "Logs": A=Waktu, B=Nama, C=Tipe Pelanggaran, D=Kode Ujian // Sheet "Status": A=Kode Ujian, B=Status (active/stopped), C=Waktu Update // Sheet "Peserta": A=Kode Ujian, B=Nama, C=Status (aktif/selesai/dikunci), D=Waktu var SHEET_LOGS = 'Logs'; var SHEET_STATUS = 'Status'; var SHEET_PESERTA = 'Peserta'; function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.action === 'stop') { setStatus(ss, data.kode, 'stopped', data.waktu); } else if (data.action === 'activate') { setStatus(ss, data.kode, 'active', data.waktu); clearPeserta(ss, data.kode); } else if (data.action === 'join') { setPeserta(ss, data.kode, data.nama, data.status || 'aktif', data.waktu); } else if (data.action === 'leave') { setPeserta(ss, data.kode, data.nama, data.status || 'selesai', data.waktu); } else { var sh = ss.getSheetByName(SHEET_LOGS); if (!sh) { sh = ss.insertSheet(SHEET_LOGS); } sh.appendRow([data.waktu, data.nama, data.tipe, data.kode]); } return ContentService.createTextOutput(JSON.stringify({ok: true})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({error: err.message})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var kode = e.parameter.kode; var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'peserta') { var peserta = getPeserta(ss, kode); var logs = getLogs(ss, kode); return jsonResponse({peserta: peserta, logs: logs, kode: kode}); } if (!kode) { return jsonResponse({status: 'unknown'}); } var status = getStatus(ss, kode); return jsonResponse({status: status, kode: kode}); } catch (err) { return jsonResponse({error: err.message}); } } function setStatus(ss, kode, status, waktu) { var sh = ss.getSheetByName(SHEET_STATUS); if (!sh) { sh = ss.insertSheet(SHEET_STATUS); } var data = sh.getDataRange().getValues(); var found = false; var rowsToDelete = []; for (var i = 0; i < data.length; i++) { if (String(data[i][0]).trim() === String(kode).trim()) { if (!found) { sh.getRange(i + 1, 1).setValue(kode); sh.getRange(i + 1, 2).setValue(status); sh.getRange(i + 1, 3).setValue(waktu); found = true; } else { rowsToDelete.push(i + 1); } } } for (var j = rowsToDelete.length - 1; j >= 0; j--) { sh.deleteRow(rowsToDelete[j]); } if (!found) { sh.appendRow([kode, status, waktu]); } } function getStatus(ss, kode) { var sh = ss.getSheetByName(SHEET_STATUS); if (!sh) return 'active'; var data = sh.getDataRange().getValues(); for (var i = data.length - 1; i >= 0; i--) { if (String(data[i][0]).trim() === String(kode).trim()) { return data[i][1] || 'active'; } } return 'active'; } function setPeserta(ss, kode, nama, status, waktu) { var sh = ss.getSheetByName(SHEET_PESERTA); if (!sh) { sh = ss.insertSheet(SHEET_PESERTA); } var data = sh.getDataRange().getValues(); for (var i = 0; i < data.length; i++) { if (String(data[i][0]).trim() === String(kode).trim() && String(data[i][1]).trim() === String(nama).trim()) { sh.getRange(i + 1, 3).setValue(status); sh.getRange(i + 1, 4).setValue(waktu); return; } } sh.appendRow([kode, nama, status, waktu]); } function getPeserta(ss, kode) { var sh = ss.getSheetByName(SHEET_PESERTA); if (!sh) return []; var data = sh.getDataRange().getValues(); var result = []; for (var i = 0; i < data.length; i++) { if (String(data[i][0]).trim() === String(kode).trim()) { result.push({nama: data[i][1], status: data[i][2], waktu: data[i][3]}); } } return result; } function clearPeserta(ss, kode) { var sh = ss.getSheetByName(SHEET_PESERTA); if (!sh) return; var data = sh.getDataRange().getValues(); var rowsToDelete = []; for (var i = 0; i < data.length; i++) { if (String(data[i][0]).trim() === String(kode).trim()) { rowsToDelete.push(i + 1); } } for (var j = rowsToDelete.length - 1; j >= 0; j--) { sh.deleteRow(rowsToDelete[j]); } } function getLogs(ss, kode) { var sh = ss.getSheetByName(SHEET_LOGS); if (!sh) return []; var data = sh.getDataRange().getValues(); var result = []; for (var i = 0; i < data.length; i++) { if (String(data[i][3]).trim() === String(kode).trim()) { result.push({waktu: data[i][0], nama: data[i][1], tipe: data[i][2]}); } } return result; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
